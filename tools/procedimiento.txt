1. Acceder al directorio donde está el proyecto.
cd c:/taskealo

2. Construir el proyecto.
docker-compose up --build -d

3. Configurar el maestro.
docker exec -it taskealo-db_master-1 bash
mysql -u root -p

CREATE USER 'replicator'@'%' IDENTIFIED BY '12345';
GRANT REPLICATION SLAVE ON *.* TO 'replicator'@'%';
FLUSH PRIVILEGES;

4. Obtener el estado de la replicación en el maestro
SHOW MASTER STATUS;

5. Obtener la IP del maestro
docker inspect -f '{{range .NetworkSettings.Networks}}{{.IPAddress}}{{end}}' taskealo-db_master-1

6. Configurar el esclavo 1.
docker exec -it taskealo-db_slave1-1 bash 
mysql -u root -p

STOP SLAVE;
RESET SLAVE;
CHANGE MASTER TO MASTER_HOST='', MASTER_USER='replicator', MASTER_PASSWORD='12345', MASTER_LOG_FILE='', MASTER_LOG_POS=;
START SLAVE;

7. Configurar el esclavo 2.
docker exec -it taskealo-db_slave2-1 bash 
mysql -u root -p

STOP SLAVE;
RESET SLAVE;
CHANGE MASTER TO MASTER_HOST='', MASTER_USER='replicator', MASTER_PASSWORD='12345', MASTER_LOG_FILE='', MASTER_LOG_POS=;
START SLAVE;
